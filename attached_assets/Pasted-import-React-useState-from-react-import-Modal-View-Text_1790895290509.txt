import React, { useState } from 'react';
import {
  Modal,
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  Pressable,
} from 'react-native';

interface PlateCalculatorModalProps {
  visible: boolean;
  onClose: () => void;
  initialWeight?: number;
}

interface PlateResult {
  plate: number;
  count: number;
}

export const PlateCalculatorModal: React.FC<PlateCalculatorModalProps> = ({
  visible,
  onClose,
  initialWeight = 60,
}) => {
  const [targetWeight, setTargetWeight] = useState<string>(initialWeight.toString());
  const [barWeight, setBarWeight] = useState<number>(20); // Standard Olympic Barbell

  const calculatePlates = (target: number, bar: number): PlateResult[] => {
    const availablePlates = [25, 20, 15, 10, 5, 2.5, 1.25];
    let weightPerSide = (target - bar) / 2;

    if (weightPerSide <= 0 || isNaN(weightPerSide)) return [];

    const result: PlateResult[] = [];
    for (const plate of availablePlates) {
      const count = Math.floor(weightPerSide / plate);
      if (count > 0) {
        result.push({ plate, count });
        weightPerSide -= count * plate;
      }
    }
    return result;
  };

  const parsedTarget = parseFloat(targetWeight) || 0;
  const plateBreakdown = calculatePlates(parsedTarget, barWeight);
  const isValidWeight = parsedTarget > barWeight && (parsedTarget - barWeight) % 2.5 === 0;

  return (
    <Modal
      animationType="fade"
      transparent={true}
      visible={visible}
      onRequestClose={onClose}
    >
      <Pressable style={styles.overlay} onPress={onClose}>
        <Pressable style={styles.container} onPress={(e) => e.stopPropagation()}>
          <View style={styles.header}>
            <Text style={styles.title}>Plate Calculator</Text>

            <TouchableOpacity onPress={onClose} style={styles.closeButton}>
              <Text style={styles.closeText}>✕</Text>
            </TouchableOpacity>
          </View>

          {/* Barbell Selector */}
          <View style={styles.barSelector}>
            <Text style={styles.label}>Barbell Weight</Text>

            <View style={styles.toggleGroup}>
              {[20, 15, 10].map((weight) => (
                <TouchableOpacity
                  key={weight}
                  style={[
                    styles.toggleBtn,
                    barWeight === weight && styles.toggleBtnActive,
                  ]}
                  onPress={() => setBarWeight(weight)}
                >
                  <Text
                    style={[
                      styles.toggleText,
                      barWeight === weight && styles.toggleTextActive,
                    ]}
                  >
                    {weight} kg
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
          </View>

          {/* Target Weight Input */}
          <View style={styles.inputContainer}>
            <Text style={styles.label}>Target Weight (kg)</Text>

            <TextInput
              style={styles.input}
              value={targetWeight}
              onChangeText={setTargetWeight}
              keyboardType="decimal-pad"
              placeholder="e.g. 100"
              placeholderTextColor="#71717A"
              selectTextOnFocus
            />
          </View>

          {/* Results Display */}
          <View style={styles.resultContainer}>
            <Text style={styles.sublabel}>Plates needed PER SIDE:</Text>

            {parsedTarget <= barWeight ? (
              <Text style={styles.warningText}>
                Enter a weight greater than the bar ({barWeight} kg).
              </Text>
            ) : plateBreakdown.length === 0 ? (
              <Text style={styles.warningText}>
                Unable to load exact weight with standard plates.
              </Text>
            ) : (
              <View style={styles.plateGrid}>
                {plateBreakdown.map((item) => (
                  <View key={item.plate} style={styles.plateBadge}>
                    <Text style={styles.plateCount}>{item.count}x</Text>

                    <Text style={styles.plateWeight}>{item.plate} kg</Text>
                  </View>
                ))}
              </View>
            )}
          </View>

          <TouchableOpacity style={styles.doneBtn} onPress={onClose}>
            <Text style={styles.doneBtnText}>Done</Text>
          </TouchableOpacity>
        </Pressable>
      </Pressable>
    </Modal>
  );
};
